\section*{A. Downloaded factors (used as provided)}
Source: Kenneth French Data Library \cite{kf}. Each factor is the monthly return of a long--short portfolio.

\section*{B. Variables we compute}
\begin{align}
R_t &= 100\left(\frac{P_t}{P_{t-1}}-1\right)
  && P_t:\ \text{month-end adjusted close} \\
\EX^{QQQ}_t &= R^{QQQ}_t-\RF_t
  && \text{response variable} \\
\EX^{SPY}_t &= R^{SPY}_t-\RF_t \\
\mathit{SPREAD}_t &= R^{QQQ}_t-R^{SPY}_t
  && \text{response for Q3} \\
\PT_t &= 100\left[\prod_{s=t-24}^{t-1}\left(1+\frac{\MKT_s+\RF_s}{100}\right)-1\right]
  && \text{window of \cite{dm16}} \\
\mathit{Crisis}_t &= -\,\frac{\PT_t-\overline{\PT}}{s_{\PT}}
  && \text{z-score; larger = more bearish} \\
\MKT^c_t &= \MKT_t-\overline{\MKT}
  && \text{centered market factor}
\end{align}
$\overline{x}$ and $s_x$ denote the sample mean and standard deviation. $\PT_t$ uses months up to $t-1$ only.

\section*{C. Q4 model}
\begin{equation}
\begin{split}
\EX^{QQQ}_t=\alpha&+\lambda\,\mathit{Crisis}_t+\beta_1\MKT^c_t+\delta\,(\mathit{Crisis}_t\times\MKT^c_t)\\
&+\beta_2\mathit{SMB}_t+\beta_3\mathit{HML}_t+\beta_4\mathit{RMW}_t+\beta_5\mathit{CMA}_t+\varepsilon_t,
\qquad H_0:\delta=0
\end{split}
\end{equation}
